""" JARVIS - Ovozli boshqariladigan shaxsiy yordamchi ==================================================== Mikrofon orqali buyruqlarni tinglaydi, ularni tanib oladi va tovush (nutq) orqali javob qaytaradi. O'rnatish (terminalda ishga tushiring): pip install SpeechRecognition pyttsx3 pyaudio Eslatma (Windows uchun pyaudio o'rnatishda muammo bo'lsa): pip install pipwin pipwin install pyaudio Ishga tushirish: python jarvis.py """ import speech_recognition as sr import pyttsx3 import datetime import webbrowser import sys import math import re class Jarvis: def __init__(self, name="Jarvis", language="uz-UZ", wake_word=None): self.name = name self.language = language # nutqni tanish tili (masalan "uz-UZ", "ru-RU", "en-US") self.wake_word = wake_word # masalan "jarvis" — agar kerak bo'lsa # Nutqni matnga aylantiruvchi self.recognizer = sr.Recognizer() self.microphone = sr.Microphone() # Matnni nutqqa aylantiruvchi (offline ishlaydi) self.engine = pyttsx3.init() self.engine.setProperty("rate", 175) # gapirish tezligi voices = self.engine.getProperty("voices") if voices: self.engine.setProperty("voice", voices[0].id) # Fon shovqiniga moslashish with self.microphone as source: self.recognizer.adjust_for_ambient_noise(source, duration=1) # ---------- Asosiy funksiyalar ---------- def speak(self, text: str): """Matnni ovoz orqali aytadi va konsolga ham chiqaradi.""" print(f"{self.name}: {text}") self.engine.say(text) self.engine.runAndWait() def listen(self, timeout=5, phrase_time_limit=8) -> str: """Mikrofondan tinglaydi va matnga aylantiradi.""" with self.microphone as source: print("... tinglayapman ...") try: audio = self.recognizer.listen( source, timeout=timeout, phrase_time_limit=phrase_time_limit ) except sr.WaitTimeoutError: return "" try: text = self.recognizer.recognize_google(audio, language=self.language) print(f"Siz: {text}") return text.lower() except sr.UnknownValueError: return "" except sr.RequestError: self.speak("Internet ulanishida muammo bor, buyruqni tushuna olmadim.") return "" # ---------- Buyruqlarni bajarish ---------- def handle_command(self, command: str) -> bool: """ Buyruqni tahlil qiladi va bajaradi. False qaytarsa - dastur to'xtaydi. """ if not command: return True # Chiqish if any(word in command for word in ["chiq", "to'xta", "yopil", "xayr", "stop", "exit"]): self.speak("Xayr! Yaxshi kun tilayman.") return False # Vaqtni aytish if "soat" in command and ("necha" in command or "vaqt" in command): now = datetime.datetime.now().strftime("%H:%M") self.speak(f"Hozir soat {now}") return True # Sanani aytish if "sana" in command or "bugun nechan" in command: today = datetime.datetime.now().strftime("%d.%m.%Y") self.speak(f"Bugungi sana: {today}") return True # Salomlashish if any(word in command for word in ["salom", "assalomu", "hey jarvis", "hi"]): self.speak("Salom! Sizga qanday yordam bera olaman?") return True # Hisob-kitob (masalan: "hisobla 12 ko'paytirilgan 5" yoki "2 + 2 nechiga teng") calc_result = self.try_calculate(command) if calc_result is not None: self.speak(f"Natija: {calc_result}") return True # Google'da qidirish: "google'da qidir python nima" if "qidir" in command or "izla" in command: query = re.sub(r".*(qidir|izla)", "", command).strip() if query: self.speak(f"{query} bo'yicha qidiryapman") webbrowser.open(f"https://www.google.com/search?q={query}") return True # Saytni ochish if "yout" in command: self.speak("YouTube ochyapman") webbrowser.open("https://youtube.com") return True # Hech narsa mos kelmasa self.speak("Kechirasiz, buni tushunmadim. Yana urinib ko'ring.") return True def try_calculate(self, command: str): """Oddiy matematik ifodalarni topib hisoblaydi (masalan '12 + 5', '10 * 3').""" # So'zlarni belgilarga aylantirish replacements = { "qo'shish": "+", "qoshish": "+", "plyus": "+", "ayirish": "-", "minus": "-", "ko'paytirish": "*", "kopaytirish": "*", "carpa": "*", "bo'lish": "/", "bolish": "/", } text = command for word, symbol in replacements.items(): text = text.replace(word, symbol) # Faqat raqam va operatorlardan iborat qismini ajratib olish match = re.search(r"(-?\d+(\.\d+)?\s*[\+\-\*/]\s*-?\d+(\.\d+)?)", text) if not match: return None expr = match.group(1) # Xavfsiz baholash: faqat raqamlar va + - * / ( ) belgilariga ruxsat if not re.fullmatch(r"[\d\.\+\-\*/\(\)\s]+", expr): return None try: result = eval(expr, {"__builtins__": {}}, {}) return round(result, 4) if isinstance(result, float) else result except Exception: return None # ---------- Asosiy sikl ---------- def run(self): self.speak(f"{self.name} tayyor. Sizni tinglayapman.") running = True while running: command = self.listen() running = self.handle_command(command) if __name__ == "__main__": # Tilni o'zgartirish uchun language parametrini almashtiring: # O'zbekcha -> "uz-UZ" # Ruscha -> "ru-RU" # Inglizcha -> "en-US" jarvis = Jarvis(name="Jarvis", language="uz-UZ") try: jarvis.run() except KeyboardInterrupt: print("\nDastur to'xtatildi.") sys.exit(0)
